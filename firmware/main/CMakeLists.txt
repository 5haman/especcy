idf_component_register(SRCS main.cpp lcd.cpp ../common/spectrum.cpp ../vendor/z80cpp/z80.cpp
    INCLUDE_DIRS ../common ../vendor/z80cpp
    REQUIRES esp_driver_gpio esp_driver_spi esp_timer esp_system heap log freertos)
if(CMAKE_BUILD_EARLY_EXPANSION)
    return()
endif()

set(content_dir "${CMAKE_BINARY_DIR}/content")
set(content_args --mode "${SPECCY_CONTENT}" --output-dir "${content_dir}")
if(SPECCY_CONTENT STREQUAL "dizzy")
    if(NOT SPECCY_ROM OR NOT SPECCY_SNA)
        message(FATAL_ERROR "Dizzy mode requires -DSPECCY_ROM=/path/48.rom and -DSPECCY_SNA=/path/dizzy.sna")
    endif()
    list(APPEND content_args --rom "${SPECCY_ROM}" --snapshot "${SPECCY_SNA}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${SPECCY_ROM}" "${SPECCY_SNA}")
endif()
execute_process(COMMAND "${PYTHON}" "${CMAKE_CURRENT_LIST_DIR}/../tools/content.py" ${content_args}
    RESULT_VARIABLE result ERROR_VARIABLE error)
if(NOT result EQUAL 0)
    message(FATAL_ERROR "Content validation failed: ${error}")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/../tools/content.py")
target_compile_features(${COMPONENT_LIB} PRIVATE cxx_std_14)
set_target_properties(${COMPONENT_LIB} PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS OFF)
target_add_binary_data(${COMPONENT_LIB} "${content_dir}/spectrum48.rom" BINARY RENAME_TO speccy_rom)
target_add_binary_data(${COMPONENT_LIB} "${content_dir}/game48.sna" BINARY RENAME_TO speccy_sna)
if(SPECCY_CONTENT STREQUAL "diagnostic")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE SPECCY_DIAGNOSTIC=1)
else()
    target_compile_definitions(${COMPONENT_LIB} PRIVATE SPECCY_DIAGNOSTIC=0)
endif()
