cmake_minimum_required(VERSION 3.16)
option(SPECCY_HOST "Build the native headless runtime and tests" OFF)
if(SPECCY_HOST)
    project(speccy_host LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 14)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_EXTENSIONS OFF)
    option(SPECCY_SANITIZE "Use address and undefined-behavior sanitizers" ON)
    if(SPECCY_SANITIZE)
        add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
        add_link_options(-fsanitize=address,undefined)
    endif()
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_library(speccy_core common/spectrum.cpp vendor/z80cpp/z80.cpp)
    target_include_directories(speccy_core PUBLIC common vendor/z80cpp)
    target_compile_options(speccy_core PRIVATE -Wall -Wextra)
    add_executable(speccy-host host/main.cpp)
    target_link_libraries(speccy-host PRIVATE speccy_core)
    target_compile_options(speccy-host PRIVATE -Wall -Wextra -Werror)
    set(content_dir "${CMAKE_BINARY_DIR}/content")
    add_custom_command(OUTPUT "${content_dir}/spectrum48.rom" "${content_dir}/game48.sna" "${content_dir}/content.json"
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/content.py" --output-dir "${content_dir}"
        DEPENDS tools/content.py VERBATIM)
    add_custom_target(diagnostic_content ALL DEPENDS "${content_dir}/spectrum48.rom" "${content_dir}/game48.sna")
    add_executable(speccy-tests tests/runtime_test.cpp)
    target_link_libraries(speccy-tests PRIVATE speccy_core)
    target_compile_options(speccy-tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_dependencies(speccy-tests diagnostic_content)
    enable_testing()
    add_test(NAME runtime COMMAND speccy-tests "${content_dir}")
    add_test(NAME host_cli COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_test.py"
        "$<TARGET_FILE:speccy-host>" "${content_dir}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/content.py")
    add_test(NAME snapshot_converter COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/converter_test.py")
    return()
endif()

set(SPECCY_CONTENT diagnostic CACHE STRING "diagnostic or dizzy")
set_property(CACHE SPECCY_CONTENT PROPERTY STRINGS diagnostic dizzy)
set(SPECCY_ROM "" CACHE FILEPATH "Local 16 KiB ROM (Dizzy mode)")
set(SPECCY_SNA "" CACHE FILEPATH "Local 48K Dizzy SNA (Dizzy mode)")
set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_SOURCE_DIR}/sdkconfig.defaults")
if(NOT DEFINED ENV{IDF_PATH})
    message(FATAL_ERROR "Export ESP-IDF v5.4.4, or configure with -DSPECCY_HOST=ON for native tests")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
set(COMPONENTS main)
project(speccy_micro)
idf_build_get_property(speccy_idf_version IDF_VER)
if(NOT speccy_idf_version STREQUAL "v5.4.4")
    message(FATAL_ERROR "This firmware requires ESP-IDF v5.4.4 (found ${speccy_idf_version})")
endif()
